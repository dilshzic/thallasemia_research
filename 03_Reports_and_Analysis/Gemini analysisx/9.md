```python
import pandas as pd

file_path = "Thalassemia_Research_Questionnaire_-_all_versions_-_labels_-_2026-02-25-07-38-45.xlsx"
df = pd.read_excel(file_path, sheet_name=0)

columns = df.columns.tolist()

# System columns to ignore
system_cols = ['start', 'end', '_id', '_uuid', '_submission_time', '_validation_status', 
               '_notes', '_status', '_submitted_by', '__version__', '_tags', 'meta/rootUuid', '_index']

questionnaire = {}
current_part = "General"

for col in columns:
    if col in system_cols:
        continue
    
    col_str = str(col).strip()
    
    # Identify Parts/Sections
    if col_str.startswith('Part'):
        current_part = col_str
        questionnaire[current_part] = {}
        continue
        
    if current_part not in questionnaire:
        questionnaire[current_part] = {}
        
    # Handle multi-select options split by '/'
    if '/' in col_str and ('Tick all that apply' in col_str or 'severe form' in col_str or 'have you' in col_str or 'form of thalassemia do you have?' in col_str or 'If not screened, what were the reasons?' in col_str):
        parts = col_str.split('/')
        q_text = parts[0].strip()
        option_text = parts[1].strip()
        
        if q_text not in questionnaire[current_part]:
             questionnaire[current_part][q_text] = []
        if option_text not in questionnaire[current_part][q_text]:
             questionnaire[current_part][q_text].append(option_text)
    else:
        # Standard question
        if col_str not in questionnaire[current_part]:
            questionnaire[current_part][col_str] = []

print("--- RAW QUESTIONNAIRE EXTRACTION ---\n")
for part, qs in questionnaire.items():
    print(f"[{part}]")
    for q, opts in qs.items():
        print(f"Q: {q}")
        for opt in opts:
            print(f"   - {opt}")
    print()

```
