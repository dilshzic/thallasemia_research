import pandas as pd
from scipy.stats import chi2_contingency, f_oneway
import numpy as np

file_path = "Thalassemia_Research_Questionnaire_-_all_versions_-_labels_-_2026-02-25-07-38-45.xlsx"
df = pd.read_excel(file_path, sheet_name=0)

# Helper for Cramer's V
def cramers_v(confusion_matrix):
    chi2 = chi2_contingency(confusion_matrix)[0]
    n = confusion_matrix.sum().sum()
    phi2 = chi2 / n
    r, k = confusion_matrix.shape
    if n <= 1: return 0
    phi2corr = max(0, phi2 - ((k-1)*(r-1))/(n-1))
    rcorr = r - ((r-1)**2)/(n-1)
    kcorr = k - ((k-1)**2)/(n-1)
    if min((kcorr-1), (rcorr-1)) == 0: return 0
    return np.sqrt(phi2corr / min((kcorr-1), (rcorr-1)))

print("--- OCCUPATION DISTRIBUTION ---")
occ_col = ' 5. Occupation:'
df[occ_col] = df[occ_col].astype(str).str.strip().replace('nan', pd.NA)
print(df[occ_col].value_counts(dropna=False))
print("\n")

# Recreate scores
q16_cols = [c for c in df.columns if '16. What are the clinical forms' in c and 'I don’t know' not in c]
q27_cols = [c for c in df.columns if '27. Problems faced by thalassemia major' in c]
df[q16_cols] = df[q16_cols].apply(pd.to_numeric, errors='coerce').fillna(0)
df[q27_cols] = df[q27_cols].apply(pd.to_numeric, errors='coerce').fillna(0)
df['Knowledge_Score'] = df[q16_cols].sum(axis=1) + df[q27_cols].sum(axis=1)

def run_chi2(col2, name):
    temp_df = df[[occ_col, col2]].dropna()
    temp_df[col2] = temp_df[col2].astype(str).str.strip()
    
    crosstab = pd.crosstab(temp_df[occ_col], temp_df[col2])
    crosstab = crosstab.loc[(crosstab.sum(axis=1) != 0), (crosstab.sum(axis=0) != 0)]
    
    if crosstab.empty or crosstab.shape[0] < 2 or crosstab.shape[1] < 2:
        return
        
    chi2, p, dof, expected = chi2_contingency(crosstab)
    v = cramers_v(crosstab.values)
    
    print(f"[{name}]")
    print(f"Chi-square: {chi2:.2f}, p-value: {p:.4f}, Cramer's V: {v:.4f}")
    if p < 0.05:
        print(">> STATISTICALLY SIGNIFICANT")
    else:
        print(">> Not significant")
    print("-" * 40)

print("--- CHI-SQUARE TESTS (OCCUPATION VS CATEGORICAL) ---")
run_chi2('7. Education Level:', 'Occupation vs Education Level')
run_chi2('31. Do you accept marriage between two thalassemia carriers? ', 'Occupation vs Acceptance of Carrier Marriage')
run_chi2('33. What was your practice of partner screening? ', 'Occupation vs Partner Screening Practice')
run_chi2('36. Do your family members know your thalassemia status? ', 'Occupation vs Family Disclosure')
run_chi2('35. Do you think is it important for your family members to undergo screening?', 'Occupation vs Importance of Family Screening')

print("\n--- ANOVA TESTS (OCCUPATION VS CONTINUOUS SCORES) ---")
occs = df[occ_col].dropna().unique()
groups = [df[df[occ_col] == o]['Knowledge_Score'].dropna() for o in occs if len(df[df[occ_col] == o]) > 1]
f_stat, p_val = f_oneway(*groups)
print("[Occupation vs Knowledge Score]")
print(f"F-statistic: {f_stat:.2f}, p-value: {p_val:.4f}")
if p_val < 0.05:
    print(">> STATISTICALLY SIGNIFICANT")
else:
    print(">> Not significant")
print("-" * 40)

df['Age'] = pd.to_numeric(df['1. Age (What was your age at last birthday?): '], errors='coerce')
age_groups = [df[df[occ_col] == o]['Age'].dropna() for o in occs if len(df[df[occ_col] == o]) > 1]
f_stat_age, p_val_age = f_oneway(*age_groups)
print("[Occupation vs Age]")
print(f"F-statistic: {f_stat_age:.2f}, p-value: {p_val_age:.4f}")
if p_val_age < 0.05:
    print(">> STATISTICALLY SIGNIFICANT")
else:
    print(">> Not significant")
print("-" * 40)

### 🔬 Statistically Significant Associations (p < 0.05)

Occupation is heavily tied to age, education, baseline knowledge, and, importantly, how people approach marriage screening.

* **Result:** $F = 10.68$, $p < 0.001$

**2. Occupation vs. Baseline Knowledge Score**

* **Result:** $F = 4.33$, $p = 0.005$
* **Interpretation:** There is a **statistically significant difference** in how well different occupational groups understand the disease.
* **Government sector** employees had the highest knowledge score (averaging 4.86 points).
* **Private sector** employees scored the lowest (averaging 3.18 points).



**3. Occupation vs. Partner Screening Practice**

* **Result:** $\chi^2 = 36.43$, $p = 0.001$, Cramer's $V = 0.241$
* **Interpretation:** There is a **highly significant association**. The data shows that *Government sector* workers are the most likely to screen their partners proactively (before or after marriage). In contrast, a high proportion of *Private sector* workers (11 people) reported that they simply "Did not screen partner" at all.

**4. Occupation vs. Education Level**

* **Result:** $\chi^2 = 60.27$, $p < 0.001$, Cramer's $V = 0.293$
* **Interpretation:** There is a **highly significant association**. Predictably, education level dictates occupation placement, which trickles down to influence the knowledge scores and screening practices mentioned above.

### 📊 Not Statistically Significant (p > 0.05)

Regardless of where they work, respondents share the same social attitudes regarding family and marriage:

* **Acceptance of Carrier Marriage** ($p = 0.884$): Working in the government or private sector does not change the widespread rejection of a marriage between two thalassemia carriers.
* **Family Disclosure** ($p = 0.475$): Being employed or unemployed does not make a person more or less likely to tell their family they are a carrier.
* **Importance of Family Screening** ($p = 0.416$): All occupational groups universally agree that family cascade screening is highly important.

**Clinical Takeaway:** The "Not Employed" (student) demographic is young and has decent baseline knowledge, making them a prime target for pre-marital interventions. However, the data reveals a vulnerability in the **Private Sector**, which shows both the lowest clinical knowledge of the disease and the highest rate of completely skipping partner screening. Workplace screening camps should heavily prioritize private sector offices!
